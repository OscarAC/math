\section{Cardinality of Sets}

\subsection{Sets with Equal Cardinalities}

\begin{exercise}{A.}
    Show that the two given sets have equal cardinality by describing a bijection from one to the other. Describe your bijection with a formula (not as a table).
\end{exercise}

\begin{problems}
    \problem $\mathbb{R}$ and $(\sqrt{2},\infty)$

    \answer $f(x) = e^x + \sqrt{2}$, with inverse $f^{-1}(y) = \ln(y - \sqrt{2})$.

    \problem The set of even integers and the set of odd integers

    \answer $f(n) = n+1$, with inverse $f^{-1}(m) = m-1$.

    \problem $\mathbb{N}$ and $S = \set{\frac{\sqrt{2}}{n} : n \in \mathbb{N}}$

    \answer $f(n) = \dfrac{\sqrt{2}}{n}$. It is surjective by the definition of $S$, and injective since $\frac{\sqrt{2}}{n} = \frac{\sqrt{2}}{m}$ gives $n = m$.

    \problem $\mathbb{Z}$ and $S = \set{x \in \mathbb{R} : \sin{x} = 1}$

    \answer Since $\sin{x} = 1$ exactly when $x = \frac{\pi}{2} + 2\pi k$ for some $k \in \mathbb{Z}$, take $f(k) = \frac{\pi}{2} + 2\pi k$, with inverse $f^{-1}(x) = \dfrac{x - \frac{\pi}{2}}{2\pi}$.

    \problem $\set{0,1} \times \mathbb{N}$ and $\mathbb{Z}$

    \answer $f(a,b) = a - 2ab + b$, shown bijective in Exercise 12 of Section 12.2.

    \problem $\mathbb{N}$ and $\mathbb{Z}$ (Suggestion: use Exercise 18 of Section 12.2.)

    \answer $f(n) = \dfrac{(-1)^n(2n-1)+1}{4}$, shown bijective in Exercise 18 of Section 12.2.

    \problem $\mathbb{N} \times \mathbb{N}$ and $\set{(n,m) \in \mathbb{N} \times \mathbb{N} : n \leq m}$

    \answer $f(n,m) = (n,\; n+m-1)$, with inverse $f^{-1}(a,b) = (a,\; b-a+1)$.
\end{problems}

\begin{exercise}{B.}
    Answer the following questions concerning bijections from this section.
\end{exercise}

\begin{problems}[from=16]
    \problem Verify that the function $f$ in Example 13.3 is a bijection.

    \answer Here $f : (0,\infty) \to (0,1)$ is given by $f(x) = \dfrac{x}{x+1}$.

    It is injective: $\dfrac{x}{x+1} = \dfrac{y}{y+1}$ gives $x(y+1) = y(x+1)$, that is, $xy + x = xy + y$, so $x = y$.

    It is surjective: given $y \in (0,1)$, the number $x = \dfrac{y}{1-y}$ lies in $(0,\infty)$ and satisfies $f(x) = y$.
\end{problems}

\subsection{Countable and Uncountable Sets}

\begin{problems}
    \problem Prove that the set $A = \set{(m,n) \in \mathbb{N} \times \mathbb{N} : m \leq n}$ is countably infinite.

    \answer Since $\mathbb{N}$ is countably infinite, Theorem 13.5 gives that $\mathbb{N} \times \mathbb{N}$ is countably infinite. By Exercise 14 of Section 13.1 the map $f(m,n) = (m,\; m+n-1)$ is a bijection $\mathbb{N} \times \mathbb{N} \to A$, so $\abs{\mathbb{N} \times \mathbb{N}} = \abs{A}$. Therefore $\abs{\mathbb{N}} = \abs{A}$ and $A$ is countably infinite.

    \problem Prove that the set of all irrational numbers is uncountable. (Suggestion: Consider proof by contradiction using Theorems 13.4 and 13.6.)
    \begin{proof}
        Let $I = \mathbb{R} - \mathbb{Q}$ be the set of irrational numbers. First note that $I$ is infinite, since the numbers $\sqrt{2} + k$ for $k \in \mathbb{N}$ are distinct and all irrational.

        Suppose for the sake of contradiction that $I$ is not uncountable. Being infinite, $I$ must then be countably infinite. By Theorem 13.4 the set $\mathbb{Q}$ is countably infinite, so Theorem 13.6 gives that $\mathbb{Q} \cup I$ is countably infinite. But $\mathbb{Q} \cup I = \mathbb{R}$, so $\abs{\mathbb{N}} = \abs{\mathbb{R}}$, contradicting Theorem 13.2.

        Therefore $I$ is uncountable.
    \end{proof}

    \problem Prove or disprove: There exists a bijective function $f : \mathbb{Q} \to \mathbb{R}$.

    \answer This is false. By Theorem 13.4 the set $\mathbb{Q}$ is countably infinite, so $\abs{\mathbb{N}} = \abs{\mathbb{Q}}$. If a bijection $f : \mathbb{Q} \to \mathbb{R}$ existed we would have $\abs{\mathbb{Q}} = \abs{\mathbb{R}}$, and hence $\abs{\mathbb{N}} = \abs{\mathbb{R}}$, contradicting Theorem 13.2. So no such bijection exists.

    \problem Prove or disprove: The set $\mathbb{Z} \times \mathbb{Q}$ is countably infinite.

    \answer This is true. By Exercise 12 of Section 13.1 the set $\mathbb{Z}$ is countably infinite, and by Theorem 13.4 so is $\mathbb{Q}$. Theorem 13.5 then gives that $\mathbb{Z} \times \mathbb{Q}$ is countably infinite.

    \problem Prove or disprove: The set $A = \set{\frac{\sqrt{2}}{n} : n \in \mathbb{N}}$ is countably infinite.

    \answer This is true. By Exercise 6 of Section 13.1 the map $f(n) = \frac{\sqrt{2}}{n}$ is a bijection $\mathbb{N} \to A$, so $\abs{\mathbb{N}} = \abs{A}$ and $A$ is countably infinite.

    \problem Describe a partition of $\mathbb{N}$ that divides $\mathbb{N}$ into $\aleph_0$ countably infinite subsets.

    \answer For each $k \in \mathbb{N}$ let $X_k = \set{2^{k-1}(2m-1) : m \in \mathbb{N}}$ the set of naturals divisible by exactly $k-1$ factors of 2. The first few are
    \begin{align*}
        X_1 &= \set{1,3,5,7,9,\dots},\\
        X_2 &= \set{2,6,10,14,18,\dots},\\
        X_3 &= \set{4,12,20,28,36,\dots}.
    \end{align*}
    Then $P = \set{X_k : k \in \mathbb{N}}$ is the required partition. Every $n \in \mathbb{N}$ can be written as $n = 2^a(2m-1)$ for a unique pair of integers $a \geq 0$ and $m \geq 1$, obtained by dividing out every factor of 2, so $n$ lies in exactly one $X_k$, namely $k = a+1$. Each $X_k$ is countably infinite, since $m \mapsto 2^{k-1}(2m-1)$ is a bijection $\mathbb{N} \to X_k$. Finally there is one such set for each $k \in \mathbb{N}$, and these are distinct, so $P$ has $\aleph_0$ elements.

    \problem Suppose $A = \set{(m,n) \in \mathbb{N} \times \mathbb{R} : n = \pi m}$. Is it true that $\abs{\mathbb{N}} = \abs{A}$?

    \answer Yes. The condition $n = \pi m$ determines $n$ from $m$, so $A = \set{(m, \pi m) : m \in \mathbb{N}}$. The map $f : \mathbb{N} \to A$ given by $f(m) = (m, \pi m)$ is surjective by this description, and injective because $f(m) = f(m')$ forces $m = m'$ in the first coordinate. Hence $\abs{\mathbb{N}} = \abs{A}$.
\end{problems}

\subsection{Comparing Cardinalities}

\begin{problems}
    \problem Prove that the set $\mathbb{C}$ of complex numbers is uncountable.
    
    \begin{proof}
        Every real number $x$ is the complex number $x + 0i$, so $\mathbb{R} \subseteq \mathbb{C}$. The set $\mathbb{R}$ is infinite, and $\abs{\mathbb{N}} \neq \abs{\mathbb{R}}$ by Theorem 13.2, so $\mathbb{R}$ is uncountable. By Theorem 13.9, any set containing an uncountable subset is uncountable, so $\mathbb{C}$ is uncountable.
    \end{proof}

    \problem Prove or disprove: If $A \subseteq B \subseteq C$ and $A$ and $C$ are countably infinite, then $B$ is countably infinite.

    \answer This is true. Since $A$ is countably infinite it is infinite, and $A \subseteq B$, so $B$ is infinite. Thus $B$ is an infinite subset of the countably infinite set $C$, and Theorem 13.8 gives that $B$ is countably infinite.

    \problem Prove or disprove: Every infinite set is a subset of a countably infinite set.

    \answer This is false. For a counterexample take $\mathbb{R}$, which is infinite. Suppose $\mathbb{R} \subseteq S$ for some countably infinite set $S$. Then $\mathbb{R}$ is an infinite subset of $S$, so Theorem 13.8 would make $\mathbb{R}$ countably infinite, contradicting Theorem 13.2. So $\mathbb{R}$ is not a subset of any countably infinite set.

    \problem Prove or disprove: The set $\set{(a_1,a_2,a_3,\dots) : a_i \in \mathbb{Z}}$ of infinite sequences of integers is countably infinite.

    \answer This is false; the set is uncountable. Write $S$ for this set, and note $S$ is infinite, as it contains the constant sequence $(k,k,k,\dots)$ for every $k \in \mathbb{Z}$.

    Suppose for the sake of contradiction that $S$ is countably infinite. By Theorem 13.3 its elements can be arranged in an infinite list $s_1, s_2, s_3, \dots$, where we write
    \[
        s_k = (a_{k1},\; a_{k2},\; a_{k3},\; \dots).
    \]
    Now define a new sequence $b = (b_1, b_2, b_3, \dots)$ by setting $b_i = a_{ii} + 1$. Each $b_i$ is an integer, so $b \in S$. But for every $k$ the sequences $b$ and $s_k$ differ in their $k$th entry, since $b_k = a_{kk} + 1 \neq a_{kk}$. So $b$ does not appear in the list, contradicting the assumption that the list contains every element of $S$.

    Therefore $S$ is not countably infinite, and being infinite it is uncountable.

    \problem Prove that if $A$ and $B$ are finite sets with $\abs{A} = \abs{B}$, then any surjection $f : A \to B$ is also an injection. Show this is not necessarily true if $A$ and $B$ are not finite.
    \begin{proof}
        Let $\abs{A} = \abs{B} = n$ and suppose $f : A \to B$ is surjective. Suppose for the sake of contradiction that $f$ is not injective, so there are elements $a \neq a'$ in $A$ with $f(a) = f(a')$. The set $f(A)$ has one element for each distinct value of $f$, and since $a$ and $a'$ contribute the same value, the $n$ elements of $A$ produce at most $n-1$ distinct values. Hence $\abs{f(A)} \leq n-1$. But $f$ is surjective, so $f(A) = B$ and $\abs{f(A)} = n$. This gives $n \leq n-1$, a contradiction. Therefore $f$ is injective.
    \end{proof}

    \answer This can fail when the sets are infinite. Take $A = B = \mathbb{N}$ and define $f : \mathbb{N} \to \mathbb{N}$ by
    \[
        f(n) = \begin{cases}
            1   & \text{if } n = 1,\\
            n-1 & \text{if } n \geq 2.
        \end{cases}
    \]
    Then $f$ is surjective, since any $m \in \mathbb{N}$ equals $f(m+1)$. But $f$ is not injective, because $f(1) = f(2) = 1$.
\end{problems}
