\section{Mathematical Induction}

\textbf{Prove the following statements with either induction, strong induction or proof by smallest counterexample}.

\begin{problems}
    \problem For every integer $n \in \mathbb{N}$, it follows that
    $1^2 + 2^2 + 3^2 + 4^2 + \mathellipsis + n^2 = \dfrac{n(n+1)(2n+1)}{6}$.
    \begin{proof}
        We prove the statement by mathematical induction.

        \begin{enumerate}
            \item Observe that if $n = 1$, this statement is $1^2 = \dfrac{(1)(2)(3)}{6} = 1$, which is true.

            \item Assume $1^2 + 2^2 + 3^2 + 4^2 + \mathellipsis + k^2 = \dfrac{k(k+1)(2k+1)}{6}$ for some integer $k \geq 1$. Observe that this implies the statement is true for $n = k + 1$.
                \begin{align*}
                    \MoveEqLeft 1^2 + 2^2 + 3^2 + \mathellipsis + k^2 + (k+1)^2\\
                    & = (1^2 + 2^2 + 3^2 + \mathellipsis + k^2) + (k+1)^2\\
                    & = \dfrac{k(k+1)(2k+1)}{6} + (k+1)^2\\
                    & = \dfrac{k(k+1)(2k+1)}{6} + \dfrac{6(k+1)^2}{6}\\
                    & = \dfrac{k(k+1)(2k+1)+6(k+1)(k+1)}{6}\\
                    & = \dfrac{2k^3+9k^2+13k+6}{6}\\
                    & = \dfrac{(k+1)(k+2)(2k+3)}{6}\\
                    & = \dfrac{(k+1)([k+1] + 1)(2[k+1]+1)}{6}
                \end{align*}
                Therefore $1^2 + 2^2 + 3^2 + \mathellipsis + k^2 + (k+1)^2 = \dfrac{(k+1)([k+1] + 1)(2[k+1]+1)}{6}$, which means the statement is true for $n = k + 1$.
        \end{enumerate}
        It follows by induction that the statement is true for every $n \in \mathbb{N}$.
    \end{proof}

    \problem If $n \in \mathbb{N}$, then $1\cdot2 + 2\cdot3 + 3\cdot4 + 4\cdot5 + \mathellipsis + n(n+1) = \dfrac{n(n+1)(n+2)}{3}$.
    \begin{proof}
        We prove the statement by mathematical induction.

        \begin{enumerate}
            \item Observe that if $n=1$, this statement is $1(1+1) = \dfrac{1(1+1)(1+2)}{3} = 2$, which is true.

            \item Assume $1\cdot2 + 2\cdot3 + 3\cdot4 + \mathellipsis + k(k+1) = \dfrac{k(k+1)(k+2)}{3}$ for some integer $k \geq 1$. Observe that this implies the statement is true for $n = k + 1$.
                \begin{align*}
                    \MoveEqLeft 1\cdot2 + 2\cdot3 + \mathellipsis + k(k+1) + (k+1)(k+2)\\
                    & = (1\cdot2 + 2\cdot3 + \mathellipsis + k(k+1)) + (k+1)(k+2)\\
                    & = \dfrac{k(k+1)(k+2)}{3} + (k+1)(k+2)\\
                    & = \dfrac{k(k+1)(k+2)}{3} + \dfrac{3(k+1)(k+2)}{3}\\
                    & = \dfrac{k(k^2+3k+2) + 3(k^2+3k+2)}{3}\\
                    & = \dfrac{k^3+6k^2+11k+6}{3}\\
                    & = \dfrac{(k+1)(k^2+5k+6)}{3}\\
                    & = \dfrac{(k+1)(k+2)(k+3)}{3}\\
                    & = \dfrac{(k+1)([k+1]+1)([k+1]+2)}{3}
                \end{align*}
                Therefore $1\cdot2+2\cdot3+\mathellipsis+k(k+1) + (k+1)(k+2) = \dfrac{(k+1)([k+1]+1)([k+1]+2)}{3}$, which means the statement is true for $n = k + 1$.
        \end{enumerate}
        It follows by induction that the statement is true for every $n \in \mathbb{N}$.
    \end{proof}

    \problem For every natural number $n$, it follows that $\sum\limits^{n}_{i=1}(8i-5) = 4n^2-n$.
    \begin{proof}
        We prove the statement by mathematical induction.
        \begin{enumerate}
            \item Observe that if $n=1$, then the statement is $\sum\limits^{1}_{i=1}(8i-5) = 4(1^2)-1 = 3$, which is true.

            \item Assume $\sum\limits^{k}_{i=1}(8i-5) = 4k^2-k$ for some $k \geq 1$. Then
                \begin{align*}
                    \sum\limits^{k+1}_{i=1}(8i-5) &= \bigg[\sum\limits^{k}_{i=1}(8i-5)\bigg] + \bigg[8(k+1)-5\bigg]\\
                                                  &= [4k^2-k] + [8(k+1)-5]\\
                                                  &= 4k^2-k+8k+3\\
                                                  &= 4k^2+8k+4-k-1\\
                                                  &= 4(k^2+2k+1)-(k+1)\\
                                                  &= 4(k+1)(k+1)-(k+1)\\
                                                  &= 4(k+1)^2-(k+1)
                \end{align*}
                This means the statement is true for $n = k + 1$.
        \end{enumerate}
        It follows by induction that $\sum\limits^{n}_{i=1}(8i-5) = 4n^2-n$ for every natural number $n$.
    \end{proof}

    \problem If $n \in \mathbb{N}$, then $\frac{1}{2!}+\frac{2}{3!}+\frac{3}{4!}+\mathellipsis+\frac{n}{(n+1)!} = 1 - \frac{1}{(n+1)!}$.
    \begin{proof}
        We prove the statement by mathematical induction.
        \begin{enumerate}
            \item Observe that if $n=1$, then the statement is $\frac{1}{2!} = 1 - \frac{1}{2!} = \frac{1}{2}$, which is true.

            \item Assume $\frac{1}{2!}+\frac{2}{3!}+\frac{3}{4!}+\mathellipsis+\frac{k}{(k+1)!} = 1 - \frac{1}{(k+1)!}$ for some $k \geq 1$. Then
                \begin{align*}
                    \MoveEqLeft \frac{1}{2!}+\frac{2}{3!}+\frac{3}{4!}+\mathellipsis+\frac{k}{(k+1)!} + \frac{k+1}{(k+2)!}\\
                    & = \bigg[\frac{1}{2!}+\frac{2}{3!}+\frac{3}{4!}+\mathellipsis+\frac{k}{(k+1)!}\bigg] + \frac{k+1}{(k+2)!}\\
                    & = 1 - \frac{1}{(k+1)!} + \frac{k+1}{(k+2)!}\\
                    & = 1 - \frac{k+2}{(k+2)(k+1)!} + \frac{k+1}{(k+2)!}\\
                    & = 1 - \frac{k+2}{(k+2)!} + \frac{k+1}{(k+2)!}\\
                    & = 1 - \frac{(k+2)-(k+1)}{(k+2)!}\\
                    & = 1 - \frac{1}{(k+2)!}\\
                    & = 1 - \frac{1}{([k+1]+1)!}
                \end{align*}
                This means the statement is true for $n = k+1$.
        \end{enumerate}
        It follows by induction that $\frac{1}{2!}+\frac{2}{3!}+\frac{3}{4!}+\mathellipsis+\frac{n}{(n+1)!} = 1 - \frac{1}{(n+1)!}$ for every $n \in \mathbb{N}$.
    \end{proof}

    \problem For any integer $n \geq 0$, it follows that $3 \mid (5^{2n} - 1)$.
    \begin{proof}
        We prove the statement by mathematical induction.
        \begin{enumerate}
            \item Observe that if $n=0$, then the statement is $3 \mid 0$, which is true.

            \item Assume this statement is true for some integer $n = k \geq 0$.
            That is, assume $3 \mid (5^{2k}-1)$.
            This means $5^{2k}-1 = 3a$ for some integer $a$, and from this we get $5^{2k} = 3a + 1$.
            Now observe that
            \begin{align*}
                5^{2(k+1)} - 1 &= 5^{2k+2}-1\\ 
                               &= 5^2\cdot5^{2k}-1\\ 
                               &= 5^2(3a+1)-1\\ 
                               &= 25(3a+1)-1\\ 
                               &= 25\cdot3a+25-1\\ 
                               &= 3(25a+8)
            \end{align*}
            This shows $5^{2(k+1)}-1 = 3(25a+8)$. Since $25a+8 \in \mathbb{Z}$, it follows that $3 \mid (5^{2(k+1)}-1)$.
        \end{enumerate}
        
        It follows by induction that $3 \mid (5^{2n}-1)$ for every integer $n \geq 0$.
    \end{proof}

    \problem For any integer $n \geq 0$, it follows that $9 \mid (4^{3n}+8)$.
    \begin{proof}
        We prove the statement by mathematical induction.
        \begin{enumerate}
            \item Observe that if $n=0$, then the statement is $9 \mid 9$, which is true.

            \item Assume this statement is true for some integer $n = k \geq 0$.
                That is, assume $9 \mid (4^{3k}+8)$.
                This means $4^{3k} + 8 = 9a$ for some integer $a$, and from this we get $4^{3k} = 9a-8$.
                Now observe that
                \begin{align*}
                    4^{3(k+1)}+8 &= 4^{3k+3}+8\\ 
                                  &= 4^3\cdot4^{3k}+8\\ 
                                  &= 4^3(9a-8)+8\\ 
                                  &= 64(9a-8)+8\\ 
                                  &= 64(9a)-512+8\\ 
                                  &= 9(64a-56)
                \end{align*}
                This shows $4^{3(k+1)}+8 = 9(64a-56)$. Since $64a-56 \in \mathbb{Z}$, it follows that $9 \mid (4^{3(k+1)}+8)$.
        \end{enumerate}

        It follows by induction that $9 \mid (4^{3n}+8)$ for every integer $n \geq 0$.
    \end{proof}

    \problem Suppose $a \in \mathbb{Z}$. Prove that $5 \mid 2^{n}a$ implies $5 \mid a$ for any $n \in \mathbb{N}$.
    \begin{proof}
        The proof is induction on $n$.
        \begin{enumerate}
            \item Let $n=1$. Suppose $5 \mid 2^{1}a$.
                Since 5 is prime, we have $5 \mid 2$ or $5 \mid a$.
                But $5 \nmid 2$, therefore $5 \mid a$.
            
            \item Assume this statement is true for some integer $n = k \geq 1$.
                That is, assume $5 \mid 2^{k}a$ implies $5 \mid a$.
                We must show that $5 \mid 2^{k+1}a$ implies $5 \mid a$.
                Suppose $5 \mid 2^{k+1}a$.
                We can write $2^{k+1}a = 2^{k}(2a)$.
                Therefore $5 \mid 2^{k}(2a)$.
                By the induction hypothesis, we have $5 \mid 2a$.
                By the base case, we have $5 \mid a$.
                Therefore, the statement holds for $k + 1$.
        \end{enumerate}

        It follows by induction that $5 \mid 2^{n}a$ implies $5 \mid a$ for every $n \in \mathbb{N}$.
    \end{proof}

    \problem For every natural number $n$, it follows that $2^{n} + 1 \leq 3^n$.
    \begin{proof}
        We prove the statement by induction.
        \begin{enumerate}
            \item Observe that if $n=1$, then the statement is $2^1+1 \leq 3^1$, that is, $3 \leq 3$, which is true.

            \item Assume this statement is true for some natural number $k \geq 1$.
                That is, assume $2^k+1 \leq 3^k$.
                We need to show that $2^{k+1}+1 \leq 3^{k+1}$.
                By the induction hypothesis, $2^k \leq 3^k - 1$. Observe that
                \begin{align*}
                    2^{k+1} + 1 = 2(2^k) + 1 & \leq 2(3^k - 1) + 1 = 2(3^k) - 1 && \text{(induction hypothesis)}\\
                                             & \leq 3(3^k) = 3^{k+1} && \text{(since $-1 < 3^k$)}
                \end{align*}
                This shows the statement is true for $n = k + 1$.
        \end{enumerate}

        It follows by induction that $2^n+1 \leq 3^n$ for every natural number $n$.
    \end{proof}

    \problem Suppose $A_1, A_2, \mathellipsis, A_n$ are sets in some universal set $U$, and $n \geq 2$. Prove that $\overline{A_1 \cup A_2 \cup \mathellipsis \cup A_n} = \overline{A_1} \cap \overline{A_2} \cap \mathellipsis \cap \overline{A_n}$.
    \begin{proof}
        The proof is by strong induction.
        \begin{enumerate}
            \item When $n=2$ the statement is $\overline{A_1 \cup A_2} = \overline{A_1} \cap \overline{A_2}$, which is true (see Chapter 8, Exercise 11).

            \item Let $k \geq 2$. Assume the statement is true if it involves $k$ or fewer sets. Then
                \begin{align*}
                    \MoveEqLeft \overline{A_1 \cup A_2 \cup \mathellipsis \cup A_{k-1} \cup A_{k} \cup A_{k+1}}\\
                    & = \overline{A_1 \cup A_2 \cup \mathellipsis \cup A_{k-1} \cup (A_{k} \cup A_{k+1})} && \text{(regrouping)}\\
                    & = \overline{A_1} \cap \overline{A_2} \cap \mathellipsis \cap \overline{A_{k-1}} \cap \overline{A_k \cup A_{k+1}} && \text{(inductive hypothesis, $k$ sets)}\\
                    & = \overline{A_1} \cap \overline{A_2} \cap \mathellipsis \cap \overline{A_{k-1}} \cap \overline{A_{k}} \cap \overline{A_{k+1}} && \text{(two-set case)}
                \end{align*}
            Thus the statement is true when it involves $k+1$ sets.
        \end{enumerate}
        This completes the proof by strong induction.
    \end{proof}
