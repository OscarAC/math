\section{Distribution of Products---Elements of the Theory of Rational Integral Functions}


\subsection{Exercises V}

\begin{problems}[from=1,step=1]
    \problem[1] How many terms are there in the distributed product
    \[
        (a_1+a_2)(b_1+b_2+b_3)(c_1+c_2+c_3+c_4)(d_1+d_2+d_3+d_4+d_5)?
    \]
    \answer $2 \times 3 \times 4 \times 5 = 120$ terms

    \problem[6] Distribute, condense, and arrange---
    \[
        (b+c)(c+a)(a+b)(b-c)(c-a)(a-b).
    \]
    \answer 
    \begin{align*}
        \MoveEqLeft
        (b+c)(c+a)(a+b)(b-c)(c-a)(a-b) \\ 
        &= (b+c)(b-c)(c+a)(c-a)(a+b)(a-b) && \text{(commutation)}\\ 
        &= (b^2-c^2)(c^2-a^2)(a^2-b^2) && \text{(by 3.)}\\ 
        &= b^2c^4-b^4c^2+c^2a^4-c^4a^2+a^2b^2-a^4b^2 && \text{(by 11.)}
    \end{align*}

    \problem[11] Write down all the quaternary products of the three letters $x$, $y$, $z$; point out how many different types they fall into, and how many products there are of each type.
    \answer $x^4, x^3y, x^3z, x^2y^2, x^2yz, x^2z^2, xy^3, xy^2z, xyz^2, xz^3, y^4, y^3z, y^2z^2, yz^3, z^4$

    These terms fall into four types;
    of the type $x^4$ there are three products, $x^4$, $y^4$, $z^4$; of the type $x^3y$ six, $x^3y$, $x^3z$, $xy^3$, $xz^3$, $y^3z$, $yz^3$; of the type $x^2y^2$ three, $x^2y^2$, $x^2z^2$, $y^2z^2$; and of the type $x^2yz$ three, $x^2yz$, $xy^2z$, $xyz^2$.

    \problem[13] Find the sum of the coefficients in the expansion of $(2a+3b+4c)^3$.
    \answer $(2+3+4)^3 = 9^3 = 729$

    \problem[17] Distribute and condense, arranging terms of the same type together---
    \[
        (b+c)(y+z) + (c+a)(z+x) + (a+b)(x+y) - (a+b+c)(x+y+z).
    \]
    \answer 
    \begin{align*}
        \MoveEqLeft
        (b+c)(y+z) + (c+a)(z+x) + (a+b)(x+y) - (a+b+c)(x+y+z) \\ 
        &=\left[(b+c)(y+z) + (c+a)(z+x) + (a+b)(x+y)\right] - \left[(a+b+c)(x+y+z)\right] \\ 
        &= \left[2by+2cz+2ax+bz+cy+cx+az+ay+bx\right] \\ 
        & \quad - \left[ ax+ay+az+bx+by+bz+cx+cy+cz \right]\\ 
        &= 2by+2cz+2ax+bz+cy+cx+az+ay+bx\\ 
        & \quad -ax-ay-az-bx-by-bz-cx-cy-cz\\ 
        &= by+cz+ax
    \end{align*}
\end{problems}

\subsection{Exercises VI}

\begin{problems}[from=1,step=1]
    \problem[1] The variables being $x$, $y$, $z$, point out the integral functions among the following, and state their degree---
    \begin{parts}
        \item $3x^2 + 2xy + 3y^2$;
        \item $\dfrac{3}{x^2} + \dfrac{2}{xy} + \dfrac{3}{y^2}$;
        \item $x^2yz + y^2zx + z^2xy + x^3 + y^3 + z^3$;
        \item $\dfrac{x^2y^3z^2}{x^2yz} + \dfrac{x^2y^3z^2}{xy^2z} + \dfrac{x^2y^3z^2}{xyz^2}$.
    \end{parts}

    \problem[8] Distribute, and arrange according to powers of $x$---
    \[
        \{ax+(b-c)y\}\{bx+(c-a)y\}\{cx+(a-b)y\};
    \]
    and show that the sum of the coefficients of $x^2y$ and $y^3$ is zero.

    \problem[10] Show that
    \begin{multline*}
        \left(x+\frac{q}{r}y\right)\left(x+\frac{r}{p}y\right)\left(x+\frac{p}{q}y\right)
        - \left(x+\frac{r}{q}y\right)\left(x+\frac{p}{r}y\right)\left(x+\frac{q}{p}y\right)\\
        = \frac{xy(x-y)(q-r)(r-p)(p-q)}{pqr}.
    \end{multline*}

    \problem[26] Calculate the coefficient of $x^4$ in the expansion of $(1+x+x^2)^8$.

    \problem[29] Show that
    \begin{align*}
        {}_n\mathrm{C}_1 + {}_n\mathrm{C}_2 + {}_n\mathrm{C}_3 + \dotsb + {}_n\mathrm{C}_n &= 2^n - 1;\\
        1 + {}_n\mathrm{C}_2 + {}_n\mathrm{C}_4 + \dotsb &= {}_n\mathrm{C}_1 + {}_n\mathrm{C}_3 + {}_n\mathrm{C}_5 + \dotsb;\\
        {}_n\mathrm{C}_r &= {}_{n-2}\mathrm{C}_r + 2\,{}_{n-2}\mathrm{C}_{r-1} + {}_{n-2}\mathrm{C}_{r-2}.
    \end{align*}

    \problem[30] There are five boxes each containing five counters marked with the numbers 0, 1, 2, 3, 4; a counter is drawn from each of the boxes and the numbers drawn are added together. In how many different ways can the drawing be made so that the sum of the numbers shall be 8?
\end{problems}

\subsection{Exercises VIII}

\begin{problems}[from=1,step=1]
    \problem[1] Write down the most general rational integral symmetrical function of $x$, $y$, $z$, $u$ of the 3rd degree.

    \problem[3] Construct a homogeneous integral function of $x$ and $y$ of the 1st degree which shall vanish when $x=y$, and become 1 when $x=5$ and $y=2$.

    \problem[6] If $\mathrm{A}(x-3)(x-5) + \mathrm{B}(x-5)(x-7) + \mathrm{C}(x-7)(x-3) = 8x-120$ for all values of $x$, determine the coefficients $\mathrm{A}$, $\mathrm{B}$, $\mathrm{C}$.

    \problem[9] Find a rational integral function of $x$ of the 3rd degree which shall have the values $\mathrm{P}$, $\mathrm{Q}$, $\mathrm{R}$, $\mathrm{S}$ when $x=a$, $x=b$, $x=c$, $x=d$ respectively.

    \problem[19] Prove the identity---
    \[
        \Sigma(x+y-z)\left\{(y-z)^2 - (z-x)(x-y)\right\} = \Sigma x^3 - 3xyz.
    \]

    \problem[21] Show that
    \[
        (x^3+y^3+z^3-3xyz)^2 = \mathrm{X}^3 + \mathrm{Y}^3 + \mathrm{Z}^3 - 3\mathrm{XYZ},
        \quad\text{where}\quad \mathrm{X} = x^2+2yz, \text{ \&c.}
    \]
\end{problems}
