\subsection{Exercises II}

\begin{problems}[from=1,step=1]
    \problem Simplify---
    \[
        \frac{5^7 \times 12^4 \times 32^2 \times (3^2 \times 4^2 \times 5)^2}{(3 \times 15 \times 2^3)^{10}}.
    \]
    \answer
    First, we simplify the numerator,
    \begin{align*}
        &5^7 \times 12^4 \times 32^2 \times (3^2 \times 4^2 \times 5)^2 \\ 
        &= 5^7 \times (2\times2\times3)^4 \times (2^5)^2 \times (3^2 \times (2\times2)^2 \times 5)^2 && \text{(arithmetical factorization)}\\ 
        &= 5^7 \times 2^4 \times 2^4 \times 3^4 \times 2^{10} \times 3^4 \times 2^4 \times 2^4 \times 5^2 && \text{($(ab)^n = a^nb^n$; $(a^m)^n = a^{mn}$)}\\ 
        &= 5^9 \times 3^8 \times 2^{26} && \text{(commutation; $a^ma^n = a^{m+n}$)}
    \end{align*}

    Next, we simplify the denominator,
    \begin{align*}
        &(3 \times 15 \times 2^3)^{10} \\ 
        &= 3^{10} \times 15^{10} \times 2^{30} && \text{($(ab)^n = a^nb^n$; $(a^m)^n = a^{mn}$)}\\ 
        &= 3^{10} \times (3\times5)^{10} \times 2^{30} && \text{(arithmetical factorization)}\\ 
        &= 3^{10} \times 3^{10} \times 5^{10} \times 2^{30} && \text{($(ab)^n = a^nb^n$)}\\ 
        &= 3^{20} \times 5^{10} \times 2^{30} && \text{($a^ma^n = a^{m+n}$)}
    \end{align*}

    Finally, we simplify the fraction,
    \begin{align*}
        &\frac{5^9 \times 3^8 \times 2^{26}}{3^{20} \times 5^{10} \times 2^{30}} \\ 
        &= \frac{1}{3^{12} \times 2^4 \times 5} && \text{($\tfrac{a^m}{a^n} = \tfrac{1}{a^{n-m}}; n>m$)}\\ 
        &= \frac{1}{42515280}
    \end{align*}

    \problem Simplify---
    \[
        \frac{36^2 a^7 b^5 c^3 d^2}{81 a^4 b^3 c^3}.
    \]

    \problem Express in its simplest form---
    \[
        \left(\frac{a^4 c^3 y^2}{a^3 b^2 c^3 x^2 y^2}\right)^{2} \times \left(\frac{a^3 b^3 x^3}{a^3 b^2 c^3 x^3 y^2}\right)^{3} \times \left(\frac{b^2 c^4 x^2 y}{a^4 b^3 x^3 y^3}\right)^{4}.
    \]

    \problem Simplify---
    \[
        \left(\frac{45 a^3 b^2 c^4}{27 a^2 b^2 c}\right)^{2} \times \left(\frac{243 a^4 b^4 c^4 x}{180 a^2 b c}\right)^{2}.
    \]

    \problem Simplify---
    \[
        (3xyz)^2 \times \frac{x^6 y^9}{x^9 y^6} \div \left(\frac{3 x^2 z^2}{y^2 z^2}\right)^{2}.
    \]

    \problem Simplify---
    \[
        \frac{(x^3 y^2 z^2)^7 \times (y^3 z^2 x^2)^7 \times (z^3 x^2 y^2)^7}{\left(\dfrac{x^2}{yz}\right)^{7} \times \left(\dfrac{y^2}{zx}\right)^{7} \times \left(\dfrac{z^2}{xy}\right)^{7}}.
    \]

    \problem Simplify---
    \[
        \left(\frac{x^m}{x^n}\right)^{m+n} \times \left(\frac{x^n}{x^l}\right)^{n+l} \times \left(\frac{x^l}{x^m}\right)^{l+m}.
    \]

    \problem Simplify---
    \[
        \left\{ \frac{(a^{p-q})^{q} \times (a^{q-r})^{r-p}}{(a^{r+q})^{r-q}} \right\}^{1/r}.
    \]

    \problem Simplify---
    \[
        \frac{(x^{a-b} \times x^{b-c})^{a} \times \left(\dfrac{x^a}{x^c}\right)}{(x^a \times x^c)^{a} \div (x^{a+c})^{c}}.
    \]

    \problem Simplify---
    \[
        \left(\frac{x^p}{x^q}\right)^{p+q} \div \left(\frac{x^{p+q}}{x^{p-q}}\right)^{p^2/q}.
    \]

    \problem Simplify---
    \[
        \left\{ \left(\frac{x^l}{x^m}\right)^{l} \times \left(\frac{x^m}{x^l}\right)^{m} \right\} \div \{(x^l)^l \times (x^m)^m\} \times \{(x^m)^l \times (x^l)^m\}.
    \]

    \problem Prove that---
    \[
        \frac{(yz)^{qr} (zx)^{rp} (xy)^{pq}}{(y^{q-1} z^{r-1})^{p} (z^{r-1} x^{p-1})^{q} (x^{p-1} y^{q-1})^{r}} = \frac{(xyz)^{p+q+r}}{x^p y^q z^r}.
    \]

    \problem Distribute the product---
    \[
        (x^{b-c} - x^{c-a} + x^{a-b}) \left( \frac{1}{x^{b-c}} + \frac{1}{x^{c-a}} + \frac{1}{x^{a-b}} \right).
    \]

    \problem Distribute---
    \[
        \left(a^p + \frac{1}{b^q}\right) \left(a^q + \frac{1}{b^p}\right)^{2}.
    \]

    \problem If $m = a^x$, $n = a^y$, $a^2 = (m^y n^x)^z$; show that $xyz = 1$.
